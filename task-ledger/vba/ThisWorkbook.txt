Private Sub Workbook_Open()
    On Error Resume Next
    ' ブックが開き切ってから3秒後に自動配信チェックを実行
    ' (設定シートの「自動配信」がOFFなら何もしません)
    Application.OnTime Now + TimeSerial(0, 0, 3), "modDigest.SendDigestAuto"
End Sub
